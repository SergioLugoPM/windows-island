1//! System statistics — CPU, RAM, network speed, battery, and (optional) CPU
2//! temperature via LibreHardwareMonitor's HTTP server.
3//!
4//! Architecture notes:
5//!  - We keep ONE persistent `sysinfo::System` across calls (in `Mutex<System>`),
6//!    refresh just CPU + memory + networks on each tick. Constructing a fresh
7//!    `System` per call would re-walk /proc and take ~50ms.
8//!  - CPU usage requires two samples ~200ms apart for a meaningful number.
9//!    We always have at least one prior sample because the manager runs every 1s.
10//!  - Network speed = bytes since last refresh / time since last refresh.
11//!  - Battery via `GetSystemPowerStatus` raw FFI — no extra crate needed.
12//!  - CPU temperature: best-effort GET to http://localhost:8085/data.json
13//!    (LHM default). Silent fail if not running.
14
15use serde::{Deserialize, Serialize};
16use std::sync::Mutex;
17use std::time::Instant;
18use sysinfo::{CpuRefreshKind, MemoryRefreshKind, Networks, RefreshKind, System};
19
20#[derive(Serialize, Deserialize, Clone, Debug, Default)]
21pub struct SystemStats {
22    /// 0..100 — overall CPU usage averaged across cores
23    pub cpu_percent: f32,
24    /// 0..100 — used RAM / total RAM
25    pub ram_percent: f32,
26    /// Total RAM in MiB (display only)
27    pub ram_used_mb: u64,
28    pub ram_total_mb: u64,
29    /// Aggregate network throughput in KiB/s since the last refresh
30    pub net_down_kbps: f64,
31    pub net_up_kbps: f64,
32    /// 0..100, or -1 if no battery (desktop)
33    pub battery_percent: i32,
34    pub battery_charging: bool,
35    /// CPU package temperature in °C; None if unavailable
36    pub cpu_temp_c: Option<f32>,
37}
38
39pub struct StatsState {
40    pub sys: Mutex<System>,
41    pub networks: Mutex<Networks>,
42    pub last_refresh: Mutex<Instant>,
43}
44
45impl StatsState {
46    pub fn new() -> Self {
47        let sys = System::new_with_specifics(
48            RefreshKind::new()
49                .with_cpu(CpuRefreshKind::everything())
50                .with_memory(MemoryRefreshKind::everything()),
51        );
52        let networks = Networks::new_with_refreshed_list();
53        Self {
54            sys: Mutex::new(sys),
55            networks: Mutex::new(networks),
56            last_refresh: Mutex::new(Instant::now()),
57        }
58    }
59}
60
61impl Default for StatsState {
62    fn default() -> Self { Self::new() }
63}
64
65/// Build a SystemStats snapshot using the persistent System instance.
66/// First call has a 200ms blocking pause to seed CPU deltas.
67pub fn collect(state: &StatsState) -> SystemStats {
68    let mut sys = state.sys.lock().unwrap();
69    let mut networks = state.networks.lock().unwrap();
70    let mut last = state.last_refresh.lock().unwrap();
71
72    let now = Instant::now();
73    let elapsed_s = now.duration_since(*last).as_secs_f64().max(0.001);
74    *last = now;
75
76    sys.refresh_cpu_usage();
77    sys.refresh_memory();
78    networks.refresh();
79
80    let cpu_percent = sys.global_cpu_usage();
81
82    let ram_total = sys.total_memory();
83    let ram_used  = sys.used_memory();
84    let ram_total_mb = ram_total / (1024 * 1024);
85    let ram_used_mb  = ram_used  / (1024 * 1024);
86    let ram_percent  = if ram_total > 0 {
87        (ram_used as f32 / ram_total as f32) * 100.0
88    } else { 0.0 };
89
90    let (mut rx_bytes, mut tx_bytes) = (0u64, 0u64);
91    for (_name, net) in networks.iter() {
92        rx_bytes += net.received();
93        tx_bytes += net.transmitted();
94    }
95    let net_down_kbps = (rx_bytes as f64 / elapsed_s) / 1024.0;
96    let net_up_kbps   = (tx_bytes as f64 / elapsed_s) / 1024.0;
97
98    let (battery_percent, battery_charging) = battery_status();
99    let cpu_temp_c = read_cpu_temp_lhm();
100
101    SystemStats {
102        cpu_percent,
103        ram_percent,
104        ram_used_mb,
105        ram_total_mb,
106        net_down_kbps,
107        net_up_kbps,
108        battery_percent,
109        battery_charging,
110        cpu_temp_c,
111    }
112}

114// ─── Battery via GetSystemPowerStatus ──────────────────────────────────────
115
116#[cfg(windows)]
117mod batt {
118    #[repr(C)]
119    pub struct SystemPowerStatus {
120        pub ac_line_status:        u8,
121        pub battery_flag:          u8,
122        pub battery_life_percent:  u8,
123        pub system_status_flag:    u8,
124        pub battery_life_time:     u32,
125        pub battery_full_life_time: u32,
126    }
127    #[link(name = "kernel32")]
128    extern "system" {
129        pub fn GetSystemPowerStatus(lp_status: *mut SystemPowerStatus) -> i32;
130    }
131}
132
133#[cfg(windows)]
134fn battery_status() -> (i32, bool) {
135    let mut s = batt::SystemPowerStatus {
136        ac_line_status: 0, battery_flag: 0, battery_life_percent: 0,
137        system_status_flag: 0, battery_life_time: 0, battery_full_life_time: 0,
138    };
139    let ok = unsafe { batt::GetSystemPowerStatus(&mut s) } != 0;
140    if !ok { return (-1, false); }
141    // battery_flag == 128 means "no system battery" (desktop)
142    if s.battery_flag == 128 || s.battery_life_percent > 100 { return (-1, false); }
143    let pct = s.battery_life_percent as i32;
144    let charging = s.ac_line_status == 1;
145    (pct, charging)
146}
147
148#[cfg(not(windows))]
149fn battery_status() -> (i32, bool) { (-1, false) }
